🌐 Host Discovery

Ping Sweep

# Using fping
fping -a -g 10.10.10.0/24 2>/dev/null

# Using nmap
nmap -sn 10.10.10.0/24 -oG - | grep "Up" | cut -d " " -f 2

# Using bash loop
for i in $(seq 1 254); do (ping -c 1 10.10.10.$i | grep "bytes from" &); done

🎯 Port Scanning

Nmap Quick Scans

# Quick TCP scan
nmap -sC -sV -oA nmap/initial TARGET

# Full TCP port scan
nmap -p- -sC -sV -oA nmap/full TARGET

# UDP scan (top 20)
nmap -sU --top-ports 20 -oA nmap/udp TARGET

# Aggressive scan
nmap -A -T4 TARGET

# Vulnerability scan
nmap --script vuln TARGET

Fast Scanners

# Rustscan (very fast)
rustscan -a TARGET -- -sC -sV

# Masscan
masscan -p1-65535,U:1-65535 TARGET --rate=1000 -e tun0

📂 FTP (21)

# Anonymous login
ftp TARGET

> anonymous

# Nmap scripts

# Gobuster

📁 SMB (139/445)

# Enumerate shares
smbclient -L //TARGET -N
smbmap -H TARGET
smbmap -H TARGET -u guest

# Connect to share
smbclient //TARGET/share -N
smbclient //TARGET/share -U username

# Enum4linux
enum4linux -a TARGET

# Nmap scripts
nmap --script smb-enum-shares,smb-enum-users,smb-os-discovery -p 445 TARGET
nmap --script smb-vuln* -p 445 TARGET

# CrackMapExec

crackmapexec smb TARGET -u user -p password --shares

📒 LDAP (389)

🐧 Linux Loot Hunting

History Files

# Bash history
cat ~/.bash_history
cat /home/*/.bash_history

# Other history files
cat ~/.zsh_history
cat ~/.mysql_history
cat ~/.psql_history

# Find all history files
find / -name "*history*" 2>/dev/null

Configuration & Credentials

# Search for passwords in configs
grep -ri "password" /etc/ 2>/dev/null
grep -ri "password" /var/www/ 2>/dev/null
grep -ri "pass=" . 2>/dev/null

# Config files with passwords
find / -name "*.conf" -exec grep -l "password" {} \; 2>/dev/null
find / -name "wp-config.php" 2>/dev/null
find / -name "config.php" 2>/dev/null
find / -name "settings.py" 2>/dev/null

# Database files
find / -name "*.db" -o -name "*.sqlite" -o -name "*.sqlite3" 2>/dev/null

# Backup files (often contain creds)
find / -name "*.bak" -o -name "*.old" -o -name "*.backup" 2>/dev/null

SSH Keys

# Find private keys
find / -name "id_rsa" 2>/dev/null
find / -name "id_ecdsa" 2>/dev/null
find / -name "id_ed25519" 2>/dev/null

# Authorized keys (who can login)
find / -name "authorized_keys" 2>/dev/null
cat ~/.ssh/*

Mail & Logs

# User mail
cat /var/mail/*
cat /var/spool/mail/*

# Auth logs
cat /var/log/auth.log
cat /var/log/secure

# Web logs
cat /var/log/apache2/access.log
cat /var/log/nginx/access.log

User Documents

# Text files in home dirs
find /home -type f -name "*.txt" 2>/dev/null
find /home -type f -name "notes*" 2>/dev/null
find /home -type f -name "*password*" 2>/dev/null

# Find everything interesting
find / -type f \( -name "*.conf" -o -name "*.config" -o -name "*password*" -o -name "*history*" \) 2>/dev/null | head -50

🪟 Windows Loot Hunting

History Files

# PowerShell history
type $env:APPDATA\Microsoft\Windows\PowerShell\PSReadLine\ConsoleHost_history.txt

# All users' PowerShell history
type C:\Users\*\AppData\Roaming\Microsoft\Windows\PowerShell\PSReadLine\ConsoleHost_history.txt

# CMD history (if session active)
doskey /history

Saved Credentials

# Windows credential manager
cmdkey /list

# WiFi passwords
netsh wlan show profiles
netsh wlan show profile name="SSID" key=clear

# SAM & SYSTEM (requires admin)
reg save HKLM\SAM sam.bak
reg save HKLM\SYSTEM system.bak

# Autologon credentials
reg query "HKLM\SOFTWARE\Microsoft\Windows NT\Currentversion\Winlogon" 2>nul | findstr "DefaultUserName DefaultDomainName DefaultPassword"

Unattend & Config Files

# Unattend files (may have passwords)
dir /s C:\unattend.xml 2>nul
dir /s C:\sysprep.inf 2>nul
dir /s C:\sysprep\sysprep.xml 2>nul

# IIS config
type C:\inetpub\wwwroot\web.config 2>nul
type C:\Windows\Microsoft.NET\Framework64\v4.0.30319\Config\web.config 2>nul

# GPP passwords (if domain joined)
findstr /S /I cpassword \\domain\sysvol\*.xml 2>nul

Search for Passwords

# Search in common file types
findstr /si "password" *.txt *.xml *.config *.ini
findstr /spin "password" *.*

# Recursive search
findstr /s /i "password" C:\Users\*.txt C:\Users\*.xml C:\Users\*.config 2>nul

# Search for specific strings
findstr /si "pwd=" *.* 2>nul
findstr /si "credential" *.* 2>nul

User Documents & Files

# Desktop and Documents
dir /s C:\Users\*\Desktop\*.txt 2>nul
dir /s C:\Users\*\Documents\*.txt 2>nul
dir /s C:\Users\*\Downloads\ 2>nul

# Sticky notes database
dir "C:\Users\*\AppData\Local\Packages\Microsoft.MicrosoftStickyNotes_*\LocalState\plum.sqlite" 2>nul

# KeePass databases
dir /s *.kdbx 2>nul

# Recent documents
dir C:\Users\*\AppData\Roaming\Microsoft\Windows\Recent\ 2>nul

Browser Credentials

# Chrome Login Data
dir "C:\Users\*\AppData\Local\Google\Chrome\User Data\Default\Login Data" 2>nul

# Firefox logins
dir "C:\Users\*\AppData\Roaming\Mozilla\Firefox\Profiles\*\logins.json" 2>nul

# Edge
dir "C:\Users\*\AppData\Local\Microsoft\Edge\User Data\Default\Login Data" 2>nul
